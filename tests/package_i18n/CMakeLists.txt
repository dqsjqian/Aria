cmake_minimum_required(VERSION 3.20)
project(aria_i18n_package_consumer LANGUAGES CXX)
set(CMAKE_DISABLE_FIND_PACKAGE_Qt6 TRUE)
set(CMAKE_DISABLE_FIND_PACKAGE_Mira TRUE)
set(CMAKE_DISABLE_FIND_PACKAGE_OpenSSL TRUE)
set(CMAKE_DISABLE_FIND_PACKAGE_nlohmann_json TRUE)
find_package(aria CONFIG REQUIRED COMPONENTS i18n)
add_executable(aria_i18n_package_consumer main.cpp)
target_link_libraries(aria_i18n_package_consumer PRIVATE aria::i18n)
enable_testing()
add_test(NAME i18n_package_consumer COMMAND aria_i18n_package_consumer)
