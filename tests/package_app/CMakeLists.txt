cmake_minimum_required(VERSION 3.20)
project(aria_app_package_consumer LANGUAGES CXX)
set(CMAKE_DISABLE_FIND_PACKAGE_Qt6 TRUE)
set(CMAKE_DISABLE_FIND_PACKAGE_Mira TRUE)
set(CMAKE_DISABLE_FIND_PACKAGE_OpenSSL TRUE)
set(CMAKE_DISABLE_FIND_PACKAGE_nlohmann_json TRUE)
find_package(aria CONFIG REQUIRED COMPONENTS app)
add_executable(aria_app_package_consumer main.cpp)
target_link_libraries(aria_app_package_consumer PRIVATE aria::app)
enable_testing()
add_test(NAME app_package_consumer COMMAND aria_app_package_consumer)
